using System;using System.Linq;using System.Reflection;using System.Threading.Tasks;using UnityEngine;using UnityEditor;using Olca.Application.Dungeon;using Olca.Core.Logic.Integration;using Olca.Core.Logic.Battle;
public static class Stage11VisualProof {
 static readonly BindingFlags F=BindingFlags.Instance|BindingFlags.NonPublic;
 static DungeonBattleController C()=>UnityEngine.Object.FindObjectsByType<DungeonBattleController>(FindObjectsSortMode.None).Single();
 static BattleSystem B()=>((DungeonBattleManager)typeof(DungeonBattleController).GetField("manager",F).GetValue(C())).CurrentBattle;
 public static object Observe()=>new{view=UnityEngine.Object.FindObjectsByType<Olca.Presentation.CustomScreen.CustomScreenView>(FindObjectsInactive.Include,FindObjectsSortMode.None).Select(v=>new{active=v.isActiveAndEnabled,visible=typeof(Olca.Presentation.CustomScreen.CustomScreenView).GetField("isScreenVisible",F).GetValue(v)}).ToArray(),inputBlocked=Olca.Unity.GameplayPause.InputBlocked,editorPaused=EditorApplication.isPaused,custom=B().CustomScreen.IsOpen,active=B().Transformations.ActiveId,B().Transformations.ActivationFramesRemaining,B().Transformations.ReleaseFramesRemaining,paused=C().IsUserPaused,scale=Time.timeScale,frame=B().LogicalFrame};
 public static async Task<object> Reopen(){EditorApplication.isPaused=false;C().SetUserPaused(false);Time.timeScale=1;B().Resume(BattlePauseReason.User);B().CustomGauge.SetGauge(CustomGaugeSystem.MAX_GAUGE);((Olca.Application.Battle.BattleSessionController)typeof(DungeonBattleController).GetField("battleSession",F).GetValue(C())).TryOpenCustomScreen();await Task.Delay(2300);if(!B().CustomScreen.IsOpen)throw new Exception("Custom did not reopen");return new{open=true,frame=B().LogicalFrame,fixtureGaugeOnly=true};}
}
